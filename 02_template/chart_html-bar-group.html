var chartHtml = new nChartHtml({
	area : '.nChart-wrap', 							// chart 그려질 영역
	direction : 'vertical', 							// 가로형(vertical) / 세로형(horison) - 기본은 가로형
	labels : ['sec01', 'sec02', 'sec03', 'sec04'],				// 하단 데이터 구분자	
	stack : {								// 적층구조로 표현 - (dataset 이 2개 이상일 경우 작동)
		enabled : false,							// 적용여부 (boolean)
		intersection : false 						// 교집합 설정여부 (기본 false) - 각 값을 따로 표현할지, 가장 큰값 기준으로 나머지 값을 그 위에 겹쳐 표현할지 여부
	},
	chart : {
		width : 500,							// 차트 영역 전체 너비 - 미지정 시 100%
		height : 200							// 차트 영역 전체 높이 - 미지정 시 너비 대비 60%
	},
	title : {
		show : true,							// 타이틀 노출 여부 - 기본 true
		text : 'Chart title'						// 타이틀 텍스트 - 기본 Chart title
	},
	legend : {								// label 에 대한 안내
		show : true,							// legend 노출 여부 - 기본 true
		position : null							// 위치 (null : 상단 / 'bottom' : 하단)
	},								
	tooltip : {								// tooltip (bar 마우스 오버 시 말풍선)
		show : true, 							// 노출 여부 - 기본 true
		share : false, 							// 세트별 묶음보기 여부(boolean)부 - 기본 false
		area : null, 							// 세트별 묶음보기 - 묶음범위 (null : 기본 - 세트전체, 'group' : 세트 내 group 단위 묶음)
		tx_group : true, 						// : 제목 하단에 '그룹' 표시 여부 (boolean) - 그룹이 있는 데이터만 적용
		unit : null, 							// 값 뒤에 붙일 단위 (건/ 명 등)
		percent : false 							// 입력된 값 기준 % 로 표시할지 여부 (boolean - 기본은 false) - true 일 경우 기본값 뒤에 (00%) 형식으로 표시
		more : {								// 툴팁에 dataset 외 별도 값을 하나 추가표현할 경우
			tit : '텍스트',						// 값 타이틀
			data : [[], [], []]					// 표출 값
		}
	},
	tick : 0,								// 기준선 간격 값 지정 - 미지정 시 자동계산
	tags : ['11', '22', '33', '44'], 		// 별도 태그 텍스트 표시 (label 별로 bar 옆(아래)에 별도 text 추가 시) - group 는 제외
	dataset : [								// 데이터 그룹
		{ name : 'my', data : [15, 40, 24, 33], color : '#ff4444', accent_idx : 1},	// name : 그룹명, data : 실제 데이터 배열, color : 해당 그룹 bar 컬러, accent_idx : 강조할 li index
		{ name : 'you', data : [5, 14, 30, 21], color : '#4466ff'},
		{ name : 'we', data : [12, 20, 25, 12], color : '#aacc55'},
		{ name : 'they', data : [10, 6, 21, 14], color : '#55aacc'}
	],
});

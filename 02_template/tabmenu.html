탭 메뉴

메뉴 + 컨텐츠

메뉴와 컨텐츠를 묶는 wrapping 영역 필요.

탭 버튼 : id / class="btn-tab" / aria-controls="연결 컨텐츠 영역 id" 속성 필수

탭 컨텐츠 : id / class="btn-tab" / aria-labelledby="연결 버튼의 id" 속성 필수

탭 컨텐츠 111
탭 컨텐츠 222
탭 컨텐츠 333 
<div class="tab-wrap ex1">
	<ul class="tab-menu tab-type1">
		<li class="on"><button type="button" id="tab-ex-1" aria-controls="tab-cnt1" class="btn-tab">탭메뉴 1</button></li>
		<li><button type="button" id="tab-ex-2" aria-controls="tab-cnt2" class="btn-tab">탭메뉴 2</button></li>
		<li><button type="button" id="tab-ex-3" aria-controls="tab-cnt3" class="btn-tab">탭메뉴 3</button></li>
	</ul>

	<!-- 탭 컨텐츠 1 -->
	<div class="tab-cnt" id="tab-cnt1" aria-labelledby="tab-ex-1">
		탭 컨텐츠 111
	</div>

	<!-- 탭 컨텐츠 2 -->
	<div class="tab-cnt" id="tab-cnt2" aria-labelledby="tab-ex-2">
		탭 컨텐츠 222
	</div>

	<!-- 탭 컨텐츠 3 -->
	<div class="tab-cnt" id="tab-cnt3" aria-labelledby="tab-ex-3">
		탭 컨텐츠 333
	</div>
</div>

let test_tab = new nTabMenu({
	wrap	: '.tab-wrap.ex1', 		// 탭 메뉴 + 컨텐츠를 감싸는 영역 선택자 (기본값 : .tab-wrap)
	menu	: '.tab-menu', 		// 탭 메뉴 목록(ul) 클래스 (기본값 : .tab-menu)
	btn  	: '.btn-tab',  		// 탭 메뉴 버튼 클래스 (기본값 : .btn-tab)
	cnt		: '.tab-cnt',  		// 탭 컨텐츠 클래스 (기본값 : .tab-cnt)
	click	: true,			// 탭 버튼 클릭 시 탭 이동여부 (기본 true) : false 일 경우 버튼 클릭 시 탭이동 기능 제거
	focus	: true,			// 탭 버튼 클릭 시 해당 컨텐츠로 focus 이동할지 여부 (기본 false)
	active  : function(cnt, btn){ 	// 탭 버튼 클릭 시 콜백 ( 활성화된 컨텐츠 / 버튼 dom 반환 )
		console.log(cnt, btn)
	}
});

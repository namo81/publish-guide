자체 제작 달력 플러그인 사용

width 값은 별도 지정

달력 style은 nCalendar.css 로 설정

input 및 달력 버튼 style은 layout.css 로 설정

var 변수명 = new nCalendar({
	calInp : '.cal'						// 달력 기능 적용할 input 선택자
	showType : 'button',					// 달력 호출 방식 / button - 버튼 클릭시 / input - input 에 focus / both - 버튼,input 둘다 사용	
	calTitle : '타이틀'					// 달력 영역 타이틀 text
	splitTx : '-'						// 날짜 표기 구분선
	positionSet : true,					// 달력 위치 - true : input 바로 아래 / false : 별도 설정 없음
	gapTop : 0,						// 달력 top 위치 gap
	gapLeft : 0,						// 달력 left 위치 gap
	calType : null,						// null: 일반 기본형 ,  month : 월간달력
	week : false,						// 일간달력 전용 - 주간표기 설정
	langType : 'kr',						// kr : 한글, en : 영문 (월~일 표기)
	changeMon : true,					// 달력 상단 월 변경 select 표출 여부
	changeYear : true,					// 달력 상단 연도 변경 select 표출 여부
	monthShift : false 					// 연/월 선택 시 '월간 달력' 전환 여부 - 일간달력 전용
	yearRange : '2019:2040',					// 연도 제한 '시작연도:종료연도'
	showBtnPanel : true,					// 하단 '오늘', '닫기' 버튼 영역 표출 여부
	closeBtnTx : '닫기',					// 닫기 버튼 텍스트
	todayBtnTx : '오늘',					// 오늘 버튼 텍스트
	controls : true,						// 이전달/다음달 버튼 생성 여부
	nextTx : '>',						// 다음달 버튼 텍스트
	prevTx : '<',						// 이전달 버튼 텍스트
	enabled_array : array 					// 활성화할 버튼 배열 (특정 일자만 활성화 하고자 할 경우)
	todayLimit : false,					// 오늘 기준 선택 제한
	todayGap : '0D', 					// 오늘 기준일의 gap 설정 (ex. 내일부터 제한, 5일전까지 제한 등)
	limitType : 'before',					// 제한 방향 설정 - before : 오늘 이전 날짜 선택 제한(기본값) / after : 오늘 이후 날짜 선택 제한
	limitGap : null, 					// 제한 기간 설정 - null : 기한 없음 / nY : n년 / nM : n개월 / nD : n일 - 3가지 방식 중 택1
	onModal : false,						// 팝업 띄울 때 body 스크롤 제거 관련 옵션(true 일 경우 달력 띄우면 body scroll 방지)
	inPage : false,						// 페이지 고정형 선택
	inTarget : null,						// 페이지 고정할 영역 선택
	inCalWrap : false					// 달력의 body 가 아닌 input 과 동일레벨에 위치할지 여부
});

페이지 내 고정 표시

기본 기능 설정은 모달형과 같으며, in-page 옵션만 추가 설정

var 변수명 = new nCalendar({
	inPage : false,					//	'false' : 기본/ 모달형, 'true' : 페이지형
	inTarget : null					// 달력 컨텐츠 표시할 영역 선택 (별도 지정 없을 경우 inp-cal 영역 내 표기됨)
});

기간 제한 설정 예시

  • - 날짜 제한 없을 경우 todayLimit : false
  • - 오늘 이전날짜 선택 제한 - limitType : 'before' / 오늘 이후날짜 선택 제한 - limitType : 'after'
      'before' 가 기본값
  • - todayGap / limitGap 는 '오늘'을 기준으로, 설정된 기간으로 설정 ('1Y' 일 경우 1년 후 / '-1Y' 일 경우 1년 전)
  • - todayGap - 제한 기준일 변경값
      단위는 연/월/주/일 각각 설정 가능 / 구분은 대문자 필수 / 3가지중 택1 하여 설정 / 설정시 string 로 설정
      (ex: 2년 = '2Y' / 3개월 = '3M' / 3주 = '3W' / 20일 = '20D')
      음수 지정 가능 (ex: 2년 = '-2Y')
  • - limitGap - 달력 내 선택가 능 표출 기간을 제한할 경우 사용.
      시작일 기준으로 계산됨 (오늘 or todayGap 으로 설정한 시작일 기준)
      단위는 연/월/주/일 각각 설정 가능 / 구분은 대문자 필수 / 3가지중 택1 하여 설정 / 설정시 string 로 설정
      (ex: 2년 = '2Y' / 3개월 = '3M' / 3주 = '3W' / 20일 = '20D') 
<!-- 오늘부터 선택가능 / 기간제한 없음 -->
	todayLimit : true,
	limitType : 'before' (기본값이므로 설정하지 않아도 무관함)

<!-- 오늘 이후 & 향후 1년까지 선택가능 -->
	todayLimit : true,
	limitGap :  '1Y' (or '12M')

<!-- 내일부터 & 향후 1년까지 선택가능 -->
	todayLimit : true,
	todayGap : '1D',
	limitGap :  '1Y' (or '12M')

<!-- 6개월 전부터 & 오늘까지 선택가능 -->
	todayLimit : true,
	limitType : 'after',
	limitGap : '-6M'

<!-- 6개월 전부터 & 어제까지 선택가능 -->
	todayLimit : true,
	todayGap : '-1D',
	limitType : 'after',
	limitGap :  '-6M'

<!-- 6개월 전부터 & 내일까지 선택가능 -->
	todayLimit : true,
	todayGap : '1D',
	limitType : 'after',
	limitGap :  '-6M'

<!-- 3개월 전부터 3개월 후까지 선택가능 -->
	todayLimit : true,
	todayGap : '-3M',
	limitType : 'before',
	limitGap :  '3M'

	or 

	todayLimit : true,
	todayGap : '-3M',
	limitGap :  '3M'

nLayoutBuilder v1.0.0

옵션 안내

HTML 요소 기반 레이아웃 빌더 (canvas 미사용) / 상태(JSON) 를 원본으로 두고 DOM 은 상태에서 렌더링 / IE 미지원

의존 파일 : moveable.js (이동·크기·회전·스냅), selecto.js (선택), html-to-image.js (이미지 내보내기 시), nLayoutBuilder.css / common.js 불필요


<link rel="stylesheet" href="../css/nLayoutBuilder.css" />

<script src="../js/vendor/moveable.js"></script>
<script src="../js/vendor/selecto.js"></script>
<script src="../js/vendor/html-to-image.js"></script>
<script src="../js/nLayoutBuilder_v1.0.0.js"></script>


var 이름 = new nLayoutBuilder({
	wrap		: '#layoutBuilder',	// 빌더가 생성될 영역 (css 선택자 or element) - 필수
	width		: 800,			// 캔버스 너비(px) - 기본 800
	height		: 500,			// 캔버스 높이(px) - 기본 500
	background	: '#ffffff',		// 캔버스 배경색 - 기본 '#ffffff'
	data		: null,			// 초기 데이터 (toJSON() 결과 / items 배열 / JSON 문자열)
	snap		: true,			// 스냅(가이드라인) 사용 여부 - 기본 true
	bounds		: true,			// 캔버스 밖으로 이동 제한 - 기본 true
	keyboard	: true,			// 키보드 조작 사용 여부 - 기본 true
	history		: 50,			// 실행취소 최대 단계 - 기본 50
	label		: '레이아웃 편집 영역',	// 편집 영역 aria-label
	select		: function(selectedItems, builder){},	// 선택 변경 / 선택 요소 변경 시 (드래그 중 포함)
	change		: function(items, builder){}		// 상태 변경 확정 시 (드래그 종료, 입력 완료 등)
});

메서드 안내


// 요소 추가 / 수정 / 삭제 (ids 생략 시 선택된 요소 대상)
이름.add('text', { text:'제목', x:20, y:20, style:{ fontSize:32 } });	// type : text / rect / circle / image → 추가된 item 반환
이름.addImage(file);					// File / Blob / 이미지 경로 → Promise(item) (원본 비율, 캔버스 60% 이내로 추가)
이름.update({ x:10, rotate:45, style:{ fontWeight:'bold' } }, ids);	// 위치·크기·회전·text·src·alt·style 수정
이름.remove(ids);
이름.duplicate(ids);
이름.order('front');					// 'front'(맨 앞) / 'forward'(앞으로) / 'backward'(뒤로) / 'back'(맨 뒤)

// 선택 / 텍스트 입력
이름.select(['nlb1', 'nlb2']);	이름.clearSelect();
이름.startEdit(id);		이름.endEdit();

// 상태 조회
이름.getItems();	이름.getSelected();	이름.getItem(id);	이름.getStage();
이름.setStage({ width:1080, height:1080, background:'#f5f5f5' });

// 실행취소
이름.undo();	이름.redo();	이름.canUndo();	이름.canRedo();

// 입출력
이름.toJSON();				// { version, stage, items } 반환
이름.load(data);			// toJSON() 결과 / items 배열 / JSON 문자열
이름.exportJSON('fileName');		// .json 파일 다운로드
이름.importJSON(file);			// File / JSON 문자열 / 객체 → Promise
이름.exportImage({			// → Promise(dataURL)
	type		: 'png',	// 'png' / 'jpeg' / 'svg' - 기본 'png'
	pixelRatio	: 2,		// 배율 - 기본 2
	fileName	: 'layout',	// 다운로드 파일명 - 기본 'layout'
	download	: true,		// false 일 경우 dataURL 만 반환
	imageOption	: {}		// html-to-image 옵션 추가 전달 (skipFonts, filter 등)
});

이름.destroy();				// 기능 제거

builder.update({ style:{ fontSize:30, fontWeight:'bold' } });
  • - 폰트 스타일 : Moveable 은 transform(위치·크기·회전) 전용이므로 글꼴·크기·굵기 등은 item.style 데이터 + update() 로 제어
  • - 업로드 이미지 : dataURL 로 JSON 에 포함됨 (용량이 큰 이미지는 JSON 크기도 커짐 / 서버 저장 시 src 를 업로드 경로로 교체 권장)
  • - 이미지 내보내기 : 로컬 서버(http) 환경에서 실행 권장 - file:// 에서는 웹폰트 / 외부 이미지가 누락될 수 있음, 외부 도메인 이미지는 CORS 허용 필요
  • - 편집용 선택 박스 / 가이드라인은 캔버스(.nlb_stage) 밖에 생성되므로 내보낸 이미지에 포함되지 않음

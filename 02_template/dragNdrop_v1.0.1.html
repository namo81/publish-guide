<script src="../js/nDragnDrop_v1.0.1.js"></script>


var 이름 = new nDragnDrop({
	wrap 		: '.drag-test1', 		// 기능 설정 wrap 영역 (css 선택자 or element) - 필수
	objWrap		: '.drag-list',			// wrap 내 드래그 아이템을 직접 포함하는 요소 (css 선택자 or element) - 필수
	obj  		: 'li',				// 드래그 될 아이템 (css 선택자 / objWrap 의 직계 자식만 대상) - 필수
	func 		: 'sequence',			// 리스트변경(sequence) or 드래그앤드랍(dnd) - 필수
	area 		: '.drop-area',			// 드롭 될 영역 (css 선택자) - dnd 일 경우 필수
	selector 	: '.drag-item', 		// 아이템 내 드래그 핸들 / 없을 경우 아이템 자체 (모바일에서는 핸들 사용 권장)
	gap 		: 10,				// 드래그 시작으로 판단할 이동 거리(px) - 기본 10
	keyboard 	: true,				// sequence 키보드(방향키) 순서변경 사용 여부 - 기본 true
	classDummy 	: 'dummy',			// 마우스를 따라다니는 복제 요소 클래스 - 기본 'dummy'
	classGhost 	: 'ghost',			// 드래그 중인 원본 아이템 클래스 - 기본 'ghost'
	classOver 	: 'drop-over',			// (dnd) 드롭 영역 위에 있을 때 영역 클래스 - 기본 'drop-over'
	active      : function(tgItem, newIdx, oldIdx){	// 완료 후 콜백
		// sequence : function(tgItem, newIdx, oldIdx) - 이동된 아이템, 변경 후 index, 변경 전 index
		// dnd      : function(tgItem, area) - 드래그 된 아이템, 드롭된 영역 (영역 밖일 경우 null)
	}
});

이름.dragUpdate();	// 아이템 추가/삭제 후 재설정
이름.removeDrag();	// 기능 제거
이름.reStart();		// 기능 재실행
  • - 드래그 중 ESC 키 : 드래그 취소 (원위치)
  • - 키보드 순서변경 : 핸들(없으면 아이템)에 포커스 후 ↑ / ← 앞으로, ↓ / → 뒤로 (포커스 불가 요소는 tabindex="0" 자동 부여)
  • - 아이템 내부 input / textarea / select 에서는 드래그가 시작되지 않음
